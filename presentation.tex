% 8. Model 1: ILP Formulation
% -------------------------------------------------------------
\begin{frame}{Model 1: Integer Linear Programming (ILP)}
    \begin{block}{Multi-Objective Function}
        \[
        \min \quad Z = \alpha \sum_{k} \sum_{i} P_k \cdot \widehat{S}_{i,k} \cdot x_{i,k} 
        - \beta \sum_{k} \sum_{i} \widehat{CSAT}_{i,k} \cdot x_{i,k} 
        + \gamma \sum_{i} d_i
        \]
    \end{block}
    \vspace{0.3cm}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Decision Variables:}
            \begin{itemize}
                \item $x_{i,k} \in \{0, 1\}$: Ticket $k$ assigned to agent $i$.
                \item $d_i \ge 0$: Linearized workload deviation from mean.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Core Constraints:}
            \begin{itemize}
                \item \textbf{Single Assignment:} $\sum_{i} x_{i,k} = 1 \quad \forall k$.
                \item \textbf{Capacity Limit:} $W_i + \sum_{k} x_{i,k} \le C_i \quad \forall i$.
                \item \textbf{Tier Eligibility:} $i \in \text{Tier}(k)$.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Model 2: Erlang-C Staffing Optimization}
    \begin{columns}[c]
        \begin{column}{0.48\textwidth}
            \textbf{Multi-Server Erlang-C Formulation:}
            \vspace{0.2cm}
            \begin{itemize}
                \setlength\itemsep{0.5em}
                \item Delay Probability $C(c_j, a_j)$ evaluated per tier.
                \item Queue Wait Time:
                \[ W_q^{(j)} = \frac{C(c_j, a_j)}{c_j \mu_j - \Lambda_j} \]
                \item \textbf{Bottleneck Discovery:} Baseline was understaffed at L2 ($\rho > 1$), creating uncontrolled queues.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Optimal Staffing Allocation}
                \centering
                \vspace{0.15cm}
                \small
                \begin{tabular}{lccc}
                    \toprule
                    \textbf{Tier} & \textbf{Offered Load} & \textbf{Base $c$} & \textbf{Optimal $c^*$} \\
                    \midrule
                    Tier 1 (L1) & 5.13 Erlangs & 5 & \textbf{7 agents} \\
                    Tier 2 (L2) & 4.22 Erlangs & 3 & \textbf{6 agents} \\
                    Tier 3 (L3) & 1.47 Erlangs & 1 & \textbf{3 agents} \\
                    \midrule
                    \textbf{Total} & & \textbf{9} & \textbf{16 agents} \\
                    \bottomrule
                \end{tabular}
                \vspace{0.15cm}
            \end{block}
            \vspace{0.1cm}
            \centering
            \footnotesize \textit{Reduces average queue waiting time from 12.4 hrs to 0.48 hrs.}
        \end{column}
    \end{columns}
\end{frame}
