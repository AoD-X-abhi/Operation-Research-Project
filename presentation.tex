% 8. Model 1: Integer Linear Programming (ILP) Formulation
% -----------------------------------------------------------------------------
\begin{frame}{Model 1: Integer Linear Programming (ILP) Formulation}
    \textbf{Mathematical Program for Dynamic Ticket Dispatch:}
    \vspace{0.1cm}
    \begin{block}{Multi-Objective Objective Function}
        \[
        \min \quad Z = \alpha \sum_{k \in K} \sum_{i \in I} P_k \cdot \widehat{S}_{i,k} \cdot x_{i,k} 
        - \beta \sum_{k \in K} \sum_{i \in I} \widehat{CSAT}_{i,k} \cdot x_{i,k} 
        + \gamma \sum_{i \in I} d_i
        \]
    \end{block}
    \vspace{0.1cm}
    \textbf{Decision Variables and Linearized Workload Deviations:}
    \begin{itemize}
        \item $x_{i,k} \in \{0, 1\}$: Binary indicator if ticket $k$ is assigned to resolver $i$.
        \item $d_i \ge 0$: Absolute workload deviation from the team mean workload $\bar{W}$, linearized via:
        \[
        d_i \ge \left(W_i + \sum_{k \in K} x_{i,k} \cdot \widehat{S}_{i,k}\right) - \bar{W}, \quad 
        d_i \ge \bar{W} - \left(W_i + \sum_{k \in K} x_{i,k} \cdot \widehat{S}_{i,k}\right) \quad \forall i \in I
        \]
    \end{itemize}
\end{frame}

% 11. Model 2: Erlang-C Multi-Server Staffing Optimization
% -----------------------------------------------------------------------------
\begin{frame}{Model 2: Multi-Server Staffing via Erlang-C}
    \textbf{Steady-State Multi-Server Queueing ($M/M/c_j$):}
    \begin{columns}[T]
        \begin{column}{0.50\textwidth}
            \begin{block}{Erlang-C Delay Probability Formula}
                \[
                C(c, a) = \frac{\frac{a^c}{c!(1 - \rho)}}{\sum_{k=0}^{c-1}\frac{a^k}{k!} + \frac{a^c}{c!(1 - \rho)}}, \quad \rho = \frac{a}{c} < 1
                \]
            \end{block}
            \vspace{0.1cm}
            \textbf{Expected Waiting and Cycle Times:}
            \[
            W_q^{(j)} = \frac{C(c_j, a_j)}{c_j \mu_j - \Lambda_j}, \quad 
            W^{(j)} = W_q^{(j)} + \frac{1}{\mu_j}
            \]
            Total System Cycle Time:
            \[
            E[T_{\text{sys}}] = \sum_{j=1}^3 \frac{\Lambda_j}{\Lambda_1} W^{(j)}
            \]
        \end{column}
        \begin{column}{0.50\textwidth}
            \begin{block}{Optimal Staffing Solution ($c_j^*$)}
                \centering
                \small
                \begin{tabular}{lccc}
                    \toprule
                    \textbf{Tier} & \textbf{Load ($a_j$)} & \textbf{Baseline ($c_j$)} & \textbf{Optimal ($c_j^*$)} \\
                    \midrule
                    Tier 1 (L1) & 5.13 Erlangs & 5 & \textbf{7 agents} \\
                    Tier 2 (L2) & 4.22 Erlangs & 3 & \textbf{6 agents} \\
                    Tier 3 (L3) & 1.47 Erlangs & 1 & \textbf{3 agents} \\
                    \midrule
                    \textbf{Total} & & 9 & \textbf{16 agents} \\
                    \bottomrule
                \end{tabular}
            \end{block}
            \vspace{0.15cm}
            \textbf{Impact of Re-Staffing:}
            \begin{itemize}
                \item Baseline had $\rho > 1$ at L2 and L3 causing runaway queueing!
                \item Optimal $c^*$ cuts expected queue delay from \textbf{12.4h} down to \textbf{0.48h}.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Managerial Insights \& Operational Policy}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Strategic Level (Staffing):}
            \begin{itemize}
                \item \textbf{Targeted Tier Investment:} Queueing shows Tier 2 and Tier 3 were chronically under-staffed relative to the 58.5\% escalation flow. Adding agents to L2 yields a 4x delay reduction compared to L1.
                \item \textbf{Dynamic Capacity Sizing:} Shift resolver counts based on peak-hour Poisson arrival rates $\lambda(t)$.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Tactical Level (Dispatching):}
            \begin{itemize}
                \item \textbf{Algorithmic Workload Leveling:} Enforcing $C_i$ limits prevents superstar agent burnout and reduces turnover by keeping Gini $< 0.15$.
                \item \textbf{Skill-Based Routing:} Leveraging predictive ML ($\widehat{S}_{i,k}$) ensures tickets are paired with resolvers having the lowest expected handle times.
            \end{itemize}
        \end{column}
    \end{columns}
    \vspace{0.25cm}
    \begin{block}{Key Industrial Takeaway}
        Optimal staffing without smart dispatch still leads to internal agent imbalance; smart dispatch without sufficient staffing still causes queue delays. \textbf{True operational excellence requires both.}
    \end{block}
